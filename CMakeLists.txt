cmake_minimum_required(VERSION 3.21)
project(PredictiveSafeClick VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(SAFECLICK_BUILD_TESTS "Build the SDK-free physics decision tests" OFF)
if(SAFECLICK_BUILD_TESTS)
    enable_testing()
    add_executable(predictor_tests tests/predictor_tests.cpp)
    target_include_directories(predictor_tests PRIVATE src)
    add_test(NAME predictor COMMAND predictor_tests)
else()
    if(NOT WIN32)
        message(FATAL_ERROR "Only Win64 is supported. Use -DSAFECLICK_BUILD_TESTS=ON for Linux tests.")
    endif()
    if(NOT DEFINED ENV{GEODE_SDK})
        message(FATAL_ERROR "GEODE_SDK is missing. Run geode sdk install and geode sdk install-binaries")
    endif()
    add_library(${PROJECT_NAME} SHARED src/Main.cpp)
    add_subdirectory($ENV{GEODE_SDK} "${CMAKE_CURRENT_BINARY_DIR}/geode")

    # Geode 5.11.2 / GD 2.2081 generated-binding workaround.
    # Affected binders instantiate sizeof() on incomplete state types.
    # These three unrelated generated sources are not needed by this mod.
    if(TARGET GeodeBindings)
        get_target_property(_binding_sources GeodeBindings SOURCES)
        set(_clean_sources "")
        set(_omitted 0)
        foreach(_src IN LISTS _binding_sources)
            if(_src MATCHES "(SequenceTriggerGameObject|GJBigSpriteNode|GJChestSprite)\\.cpp$")
                message(STATUS "Geode 5.11.2 workaround: omitting ${_src}")
                math(EXPR _omitted "${_omitted}+1")
            else()
                list(APPEND _clean_sources "${_src}")
            endif()
        endforeach()
        if(_omitted GREATER 0)
            set_property(TARGET GeodeBindings PROPERTY SOURCES "${_clean_sources}")
        endif()
    endif()
    setup_geode_mod(${PROJECT_NAME})
endif()
